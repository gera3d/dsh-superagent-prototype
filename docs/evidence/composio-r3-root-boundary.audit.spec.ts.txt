import {Context} from '@deepseek-ai/cordis'
import {ToolRuntime,defineTool} from '@deepseek-ai/dsh-tools'
import {expect,it} from 'vitest'
import {installComposioBoundary,assessComposioResult} from '../src/app/composio-boundary.ts'
import {startRuntime} from './fixtures/host.ts'
const name='mcp__composio__COMPOSIO_MULTI_EXECUTE_TOOL'
async function run(value:unknown){const ctx=new Context();ctx.reflect.provide('systemPrompt',{tools:()=>()=>{}});const tools=new ToolRuntime(ctx);installComposioBoundary({ctx});tools.register(defineTool({name,description:'Offline deterministic MCP-value fixture; no external side effect',parameters:{},output:{schema:{type:'object',additionalProperties:true},render:(_a,v)=>(v as {content:any[]}).content},execute:async()=>value}));return tools.execute({callId:'audit' as never,name,arguments:{},signal:new AbortController().signal})}
it('positive control classifies a nested action failure at the real tool registry',async()=>{const r=await run({isError:false,content:[{type:'text',text:JSON.stringify({successful:false,error:'Validation failed'})}]});expect(r.isError).toBe(true)})
it('does not lose the MCP error flag when a tool returns an MCP error envelope as its value',async()=>{const r=await run({isError:true,content:[{type:'text',text:'429 Too many requests; Retry-After: 60'}]});expect(r.isError).toBe(true)})
it('decorates the ordinary-user session consumption path without granting any tools',async()=>{const h=await startRuntime();try{await (h.controller.create as (o:{sessionId:string})=>Promise<void>)({sessionId:'ordinary-audit'});const agent=h.agents.get('ordinary-audit')! as any;agent.session.header={};await (h.runtime as any).ctx.parallel('agent/created',{agent});const value={isError:false,content:[{type:'text',text:JSON.stringify({successful:false,error:'Validation failed'})}]};const result=await agent.ctx.waterfall('tools/post-execute',{name,callId:'ordinary-audit',arguments:{}},{isError:false,value,content:value.content},async()=>({kind:'accept'}));expect(result.kind).toBe('block')}finally{await h.stop()}})

it('retains known successful entries when the normalized MCP result is an error with partial batch data',()=>{const ok={index:0,tool_slug:'READ_ITEM',response:{successful:true,data:{id:'kept'}}};const payload={successful:false,data:{total_count:2,error_count:1,results:[ok,{index:1,tool_slug:'WRITE_ITEM',response:{successful:false,error:'timeout after sending'}}]}};const result=assessComposioResult(name,{isError:true,structuredContent:payload,content:[{type:'text',text:JSON.stringify(payload)}]});expect(result?.succeeded).toEqual([ok])})
